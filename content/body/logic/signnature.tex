\chapter{The Signature of Geometry}

\section{Formal Signatures}
Specification of primitive symbols and arities defining a formal system.
\begin{definition}[Signature of Geometry]
A signature $\mathcal{L}_{geo}$ consists of:
\begin{itemize}
    \item a ternary relation symbol $\mathcal{B}$ (betweenness),
    \item a quaternary relation symbol $\cong$ (congruence).
\end{itemize}
\end{definition}

\section{Geometric Language}
Formal language generated from geometric primitives and relation symbols.

\section*{System A - Midpoint Geometry}
\textbf{Primitive:} a set \(\mathbf{P}\) of points and a ternary relation
\(M \subseteq \mathbf{P}^3\).  We read \(M(A,B,C)\) as ``\(B\) is the midpoint
of \(A\) and \(C\)''.

\begin{axiom}[Uniqueness]
  \(M(A,B,C) \land M(A,B',C) \;\Longrightarrow\; B = B'\).
\end{axiom}
\begin{axiom}[Symmetry]
  \(M(A,B,C) \;\Longrightarrow\; M(C,B,A)\).
\end{axiom}
\begin{axiom}[Reflexivity]
  \(M(A,A,A)\).
\end{axiom}
\begin{axiom}[Existence]
  \(\forall A,C \in \mathbf{P},\; \exists B \in \mathbf{P}\; M(A,B,C)\).
\end{axiom}
\begin{axiom}[Cancellation]
  \(M(A,B,A) \;\Longrightarrow\; A = B\).
\end{axiom}

\textbf{Primitive:} a set \(\mathbf{P}\) and a function
\(R : \mathbf{P} \times \mathbf{P} \to \mathbf{P}\).  We write
\(R_B(A)\) for the reflection of \(A\) across \(B\).

\begin{axiom}[Involutivity]
  \(R_B(R_B(A)) = A\).
\end{axiom}
\begin{axiom}[Fixed point]
  \(R_B(B) = B\).
\end{axiom}
\begin{axiom}[Transitivity of reflections]
  \(R_C(R_B(A)) = R_{R_C(B)}(R_C(A))\).
\end{axiom}
\begin{axiom}[Existence of midpoint]
  For any \(A,C\) there exists a unique \(B\) such that \(R_B(A)=C\).
\end{axiom}
\begin{axiom}[Density]
  If \(A \neq B\), then \(R_A(B) \neq B\) and the set
  \(\{R_A^{\,n}(B) \mid n \in \mathbb{Z}\}\) has no least positive element
  (formally, there exists a point between \(A\) and \(B\)).
\end{axiom}
\end{axiom}
\begin{axiom}[Existence]
  \(\forall A,C \in \mathbf{P},\; \exists B \in \mathbf{P}\; M(A,B,C)\).
\end{axiom}
\begin{axiom}[Cancellation]
  \(M(A,B,A) \;\Longrightarrow\; A = B\).
\end{axiom}
\begin{axiom}[Density of Midpoints]
  \(A \neq B \;\Longrightarrow\; \exists C\; M(A,C,B)\).
\end{axiom}

\[
\mathcal{B}(A,B,C) \;\overset{\text{def}}{=}\;
\bigl( \exists X\, (M(A,X,B) \land M(X,B,C)) \bigr) \lor (A=B \land B=C).
\]
\begin{axiom}[Density of Midpoints]
  \(A \neq B \;\Longrightarrow\; \exists C\; M(A,C,B)\).
\end{axiom}
